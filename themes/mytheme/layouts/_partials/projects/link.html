{{- $project := .project -}} {{- $language := .page.Site.Language.Lang -}} {{- $target := "" -}} {{- if reflect.IsMap $project.url -}} {{- $target = index $project.url $language -}} {{- end -}} {{- if or (not $target) (ne (printf "%T" $target) "string") -}} {{- errorf "projects: %q のurl.%sに空でないURL文字列を指定してください。" $project.id $language -}} {{- $target = "" -}} {{- end -}} {{- $external := true -}} {{- if isset $project "external" -}} {{- if ne (printf "%T" $project.external) "bool" -}}{{- errorf "projects: %q のexternalは真偽値です。" $project.id -}}{{- end -}} {{- $external = $project.external -}} {{- end -}} {{- $url := urls.Parse $target -}} {{- $href := "" -}} {{- if $external -}} {{- if or (not (in (slice "http" "https") $url.Scheme)) (not $url.Host) $url.User -}} {{- errorf "projects: %q のurlはHTTP(S)の絶対URLを指定してください。" $project.id -}} {{- else -}}{{- $href = $target -}}{{- end -}} {{- else -}} {{- if or $url.Scheme $url.Host (strings.HasPrefix $target "//") (strings.HasPrefix $target "#") (strings.Contains $target "\\") (not $target) -}} {{- errorf "projects: %q のexternal=falseではurlにサイト内ページのパスを指定してください。" $project.id -}} {{- else -}}{{- $href = relref .page $target -}}{{- end -}} {{- end -}} {{- return (dict "href" $href "external" $external) -}}
