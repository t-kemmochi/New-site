{{- $authors := .authors | default slice -}} {{- $text := "" -}} {{- if eq .language "ja" -}} {{- $text = delimit $authors "、" -}} {{- else if eq (len $authors) 1 -}} {{- $text = index $authors 0 -}} {{- else if eq (len $authors) 2 -}} {{- $text = printf "%s and %s" (index $authors 0) (index $authors 1) -}} {{- else if gt (len $authors) 2 -}} {{- $text = printf "%s, and %s" (delimit (first (sub (len $authors) 1) $authors) ", ") (index $authors (sub (len $authors) 1)) -}} {{- end -}} {{- return $text -}}
